// Calendrier des flashcards : boîtes de Leitner (1, 3, 7, 14, 30 jours), 10 nouvelles cartes par jour.
const DAY = 24 * 60 * 60 * 1000;
export const BOITES = [1, 3, 7, 14, 30];
export const NOUVELLES_PAR_JOUR = 10;
const KEY = 'cahier.cartes.v1';

function memoryStorage() {
    const m = new Map();
    return { getItem: k => (m.has(k) ? m.get(k) : null), setItem: (k, v) => m.set(k, String(v)) };
}

function defaultStorage() {
    try { if (typeof localStorage !== 'undefined') return localStorage; } catch { /* accès refusé */ }
    return memoryStorage();
}

export function jourLocal(now) {
    const d = new Date(now);
    return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
}

export function createCartes(storage = defaultStorage(), key = KEY) {
    const num = v => typeof v === 'number' && Number.isFinite(v);
    function valide(r) {
        return !!r && typeof r === 'object' && !Array.isArray(r)
            && Number.isInteger(r.boite) && r.boite >= 1 && r.boite <= BOITES.length
            && num(r.prochaine) && num(r.vue);
    }
    function load() {
        const vide = { cartes: {}, nouvelles: { jour: '', n: 0 } };
        try {
            const d = JSON.parse(storage.getItem(key));
            if (!d || typeof d !== 'object') return vide;
            const cartes = {};
            if (d.cartes && typeof d.cartes === 'object') {
                for (const [id, r] of Object.entries(d.cartes)) if (valide(r)) cartes[id] = r;
            }
            const nv = d.nouvelles;
            const nouvelles = nv && typeof nv === 'object' && typeof nv.jour === 'string' && Number.isInteger(nv.n) && nv.n >= 0
                ? { jour: nv.jour, n: nv.n } : { jour: '', n: 0 };
            return { cartes, nouvelles };
        } catch { return vide; }
    }
    function save(d) {
        try { storage.setItem(key, JSON.stringify(d)); } catch { /* stockage plein ou refusé */ }
    }

    function getCarte(id) {
        return load().cartes[id] || null;
    }

    function noter(id, ok, now = Date.now()) {
        const d = load();
        const rec = d.cartes[id];
        if (!rec) {
            const jour = jourLocal(now);
            d.nouvelles = d.nouvelles.jour === jour ? { jour, n: d.nouvelles.n + 1 } : { jour, n: 1 };
        }
        const boite = ok ? Math.min((rec ? rec.boite : 1) + 1, BOITES.length) : 1;
        d.cartes[id] = { boite, prochaine: now + BOITES[boite - 1] * DAY, vue: now };
        save(d);
    }

    function cartesDues(ids, now = Date.now()) {
        const { cartes } = load();
        return ids.filter(id => cartes[id] && cartes[id].prochaine <= now)
            .sort((a, b) => cartes[a].prochaine - cartes[b].prochaine);
    }

    function nouvellesDuJour(ids, now = Date.now(), max = NOUVELLES_PAR_JOUR) {
        const d = load();
        const deja = d.nouvelles.jour === jourLocal(now) ? d.nouvelles.n : 0;
        const reste = Math.max(0, max - deja);
        return ids.filter(id => !d.cartes[id]).slice(0, reste);
    }

    function sessionDuJour(cartes, now = Date.now(), max = NOUVELLES_PAR_JOUR) {
        const parId = new Map(cartes.map(c => [c.id, c]));
        const tous = [...parId.keys()];
        return {
            dues: cartesDues(tous, now).map(id => parId.get(id)),
            nouvelles: nouvellesDuJour(tous, now, max).map(id => parId.get(id)),
        };
    }

    return { getCarte, noter, cartesDues, nouvellesDuJour, sessionDuJour };
}

export const cartesEtat = createCartes();

// Charge les cartes de toutes les séances marquées "cartes": true ; un fichier illisible est compté et ignoré.
export async function rassemblerCartes(index, charger) {
    const cartes = [];
    let erreurs = 0;
    for (const m of index.matieres || []) {
        for (const s of m.seances || []) {
            if (!s.cartes) continue;
            try {
                const f = await charger(s.dossier);
                if (!f || !Array.isArray(f.cartes)) throw new Error('cartes.json invalide');
                for (const c of f.cartes) cartes.push({ ...c, seanceId: s.id, seanceTitre: `${s.type} · ${s.titre}` });
            } catch { erreurs++; }
        }
    }
    return { cartes, erreurs };
}
```

import { escapeHtml } from './utils.js?v=6';

// Flashcards : recto, « Voir la réponse » (Espace), puis Je ne savais pas (1) / Je savais (2). Bilan en fin de série.
export function lancerCartes(conteneur, cartes, { onNote, onFin, libre = false } = {}) {
    if (conteneur._arreterCartes) conteneur._arreterCartes();
    const total = cartes.length;
    let i = 0;
    let sues = 0;
    let retournee = false;
    const ratees = [];

    if (total === 0) {
        conteneur.innerHTML = '<p class="quiz-vide">Aucune carte pour le moment.</p>';
        return;
    }

    // Raccourcis clavier, actifs seulement quand la série est affichée et visible.
    function touche(e) {
        if (!conteneur.isConnected) { arreter(); return; }
        if (conteneur.offsetParent === null || e.ctrlKey || e.metaKey || e.altKey) return;
        if (e.target instanceof Element && e.target.closest('input, textarea, select, [contenteditable]')) return;
        if (e.key === ' ' && !retournee) { e.preventDefault(); retourner(); }
        else if (e.key === '1') noter(false);
        else if (e.key === '2') noter(true);
    }
    function arreter() {
        document.removeEventListener('keydown', touche);
        if (conteneur._arreterCartes === arreter) delete conteneur._arreterCartes;
    }
    document.addEventListener('keydown', touche);
    conteneur._arreterCartes = arreter;

    function afficher() {
        retournee = false;
        const c = cartes[i];
        conteneur.innerHTML = `
            <div class="carte">
                <div class="quiz-top"><span>Carte ${i + 1} sur ${total}</span>
                    <span class="quiz-bar"><i style="width:${(i / total) * 100}%"></i></span></div>
                ${c.seanceTitre ? `<p class="carte-seance">${escapeHtml(c.seanceTitre)}</p>` : ''}
                <p class="carte-recto">${escapeHtml(c.recto)}</p>
                <div class="carte-verso" hidden>
                    <p class="carte-reponse">${escapeHtml(c.verso)}</p>
                    <p class="quiz-source">Source : ${escapeHtml(c.source)}</p>
                </div>
                <div class="carte-actions">
                    <button class="btn btn-primary" data-action="retourner">Voir la réponse <kbd>Espace</kbd></button>
                    <button class="btn btn-secondary" data-action="non" hidden>Je ne savais pas <kbd>1</kbd></button>
                    <button class="btn btn-primary" data-action="oui" hidden>Je savais <kbd>2</kbd></button>
                </div>
            </div>`;
        conteneur.querySelector('[data-action="retourner"]').addEventListener('click', retourner);
        conteneur.querySelector('[data-action="non"]').addEventListener('click', () => noter(false));
        conteneur.querySelector('[data-action="oui"]').addEventListener('click', () => noter(true));
    }

    function retourner() {
        if (retournee || i >= total) return;
        retournee = true;
        conteneur.querySelector('.carte-verso').hidden = false;
        conteneur.querySelector('[data-action="retourner"]').hidden = true;
        conteneur.querySelectorAll('[data-action="non"], [data-action="oui"]').forEach(b => { b.hidden = false; });
    }

    function noter(ok) {
        if (!retournee || i >= total) return;
        const c = cartes[i];
        if (onNote) onNote(c, ok);
        if (ok) sues++; else ratees.push(c);
        i++;
        if (i < total) afficher(); else bilan();
    }

    function bilan() {
        arreter();
        conteneur.innerHTML = `
            <div class="carte quiz-bilan">
                <h3 class="quiz-enonce">${sues} sur ${total} sue${sues > 1 ? 's' : ''}</h3>
                <p>${ratees.length ? `${ratees.length} à revoir.` : 'Tout était su.'}</p>
                <div class="quiz-actions">
                    ${libre && ratees.length ? '<button class="btn btn-secondary" id="cartes-refaire">Refaire celles ratées</button>' : ''}
                    <button class="btn btn-primary" id="cartes-fin">Terminer</button>
                </div>
            </div>`;
        conteneur.querySelector('#cartes-fin').addEventListener('click', () => onFin && onFin());
        const refaire = conteneur.querySelector('#cartes-refaire');
        if (refaire) refaire.addEventListener('click', () => lancerCartes(conteneur, ratees, { onNote, onFin, libre }));
    }

    afficher();
}
```
